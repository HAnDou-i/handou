-- ============================================================ -- 1188188.xyz 首页访问量统计 -- 位置：Supabase Dashboard → SQL Editor，整段粘贴执行。可重复执行。 -- -- 设计要点 -- · 计数存在你自己库里，不依赖第三方统计服务 -- · 匿名（anon）只能「读总数」和「调用自增函数」，不能直接改数字 -- · 自增走 security definer 函数，绕过 RLS 但不放开表的写权限 -- ============================================================ -- ① 计数表（固定只有一行，id = 1） create table if not exists public.site_views ( id int primary key, hits bigint not null default 0, updated_at timestamptz not null default now() ); insert into public.site_views (id, hits) values (1, 0) on conflict (id) do nothing; alter table public.site_views enable row level security; -- ② 允许匿名读总数（页面上要显示） drop policy if exists site_views_read on public.site_views; create policy site_views_read on public.site_views for select to anon, authenticated using (true); -- ③ 自增函数：谁都能调，但只能 +1，改不了别的 create or replace function public.bump_view() returns bigint language plpgsql security definer set search_path = public as $$ declare n bigint; begin update public.site_views set hits = hits + 1, updated_at = now() where id = 1 returning hits into n; return coalesce(n, 0); end; $$; revoke all on function public.bump_view() from public; grant execute on function public.bump_view() to anon, authenticated; -- ④ 自检：执行后这一段应该返回一行，hits = 1 select public.bump_view() as hits_after_test;
